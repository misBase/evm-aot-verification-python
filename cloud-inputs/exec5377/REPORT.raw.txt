  ```python
  class PrefixSolver:
      def __init__(self):
          self.solver, self.stack = SolverFor("QF_AUFBV"), []
      def check(self, conjuncts, timeout_ms):
          common = 0
          for old, new in zip(self.stack, conjuncts):
              if old.get_id() != new.get_id():
                  break
              common += 1
          if len(self.stack) > common:
              self.solver.pop(len(self.stack) - common); del self.stack[common:]
          for c in conjuncts[common:]:
              self.solver.push(); self.solver.add(c); self.stack.append(c)
          return _bounded_check(self.solver, timeout_ms, "incremental")
  ```
